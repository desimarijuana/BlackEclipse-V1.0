# BlackEclipse harness runtime — Python agent service.
#
# Installed by scripts/setup-python.js (and checked by scripts/doctor.js).
# The app's agent runtime is Python + LangChain + LangGraph; the Next.js side
# owns the HTTP API, JDdS tool execution, and run/event persistence.
#
# Keep this list minimal. Provider integrations are additive: adding one is a
# line here plus a catalog entry in harness/models.py.

# Service
fastapi>=0.115,<1
uvicorn[standard]>=0.30
pydantic>=2.7

# Full JSON Schema validation of skill inputs/outputs (nested types, arrays,
# enums, bounds, required). Already present transitively via the MCP SDK; pinned
# explicitly because skill contract enforcement depends on it directly.
jsonschema>=4.21,<5

# Agent runtime
langchain>=1.0
langchain-core>=1.0
langgraph>=1.0

# MCP client (Phase 5): local stdio servers. The maintained MCP Python SDK;
# pinned <3 so a future v3 does not change the transport/lifecycle under us.
mcp>=2,<3

# The MCP SDK's HTTP transport imports `httpx2` directly (it is re-exported by
# mcp.client.streamable_http / mcp.client.sse). It is a declared transitive
# dependency of `mcp` (`httpx2>=2.5.0`), pinned here explicitly with an upper
# bound so an unrelated upgrade cannot change the transport under us. We do not
# depend on it anywhere except through the MCP SDK; the lighter alternative
# (the SDK's own client, which it builds on httpx2) is what we use.
httpx2>=2.5,<3

# Model providers (universal coverage).
# langchain-openai also powers EVERY OpenAI-compatible endpoint (Ollama, LM
# Studio, OpenRouter, DeepSeek, xAI, Perplexity, Together, Fireworks, custom).
langchain-openai>=1.0
langchain-anthropic>=0.3
langchain-google-genai>=2.0
langchain-mistralai>=0.2
langchain-groq>=0.2

# Real pseudo-terminal for the terminal subsystem's persistent shells. Optional
# in effect (the session layer falls back to a piped shell and reports
# `pty: false`), but installed by default so interactive shells, ANSI output and
# window resizing work out of the box. Windows gets a bundled ConPTY; POSIX gets
# a forkpty wrapper. Kept as platform markers so neither is installed needlessly.
pywinpty>=2,<4; sys_platform == "win32"
ptyprocess>=0.7; sys_platform != "win32"

# The Code Execution & Verification Layer's SQL runtime reaches PostgreSQL and
# DuckDB through these. SQLite needs no driver (stdlib). Both are optional in
# effect — a missing one makes its dialect REFUSE with SQL_DRIVER_MISSING rather
# than fabricate a connection — but installed by default so `execute_sql` can
# actually reach the dialects Eclipse advertises. `pg8000` is pure-Python (no
# build toolchain); `duckdb` is an embedded wheel (no server).
pg8000>=1.31,<2
duckdb>=1.0

# Tests
pytest>=8.0
pytest-asyncio>=0.24
