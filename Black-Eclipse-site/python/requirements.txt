# The MCP SDK's HTTP transport imports `httpx2` directly (it is re-exported by

# depend on it anywhere except through the MCP SDK; the lighter alternative
# (the SDK's own client, which it builds on httpx2) is what we use.

# Real pseudo-terminal for the terminal subsystem's persistent shells. Optional
# in effect (the session layer falls back to a piped shell and reports
# `pty: false`), but installed by default so interactive shells, ANSI output and
# window resizing work out of the box. Windows gets a bundled ConPTY; POSIX gets
# a forkpty wrapper. Kept as platform markers so neither is installed needlessly.
pywinpty>=2,<4; sys_platform == "win32"

# The Code Execution & Verification Layer's SQL runtime reaches PostgreSQL and
# DuckDB through these. SQLite needs no driver (stdlib). Both are optional in
# effect — a missing one makes its dialect REFUSE with SQL_DRIVER_MISSING rather
# than fabricate a connection — but installed by default so `execute_sql` can
# actually reach the dialects Eclipse advertises. `pg8000` is pure-Python (no
# build toolchain); `duckdb` is an embedded wheel (no server).
pg8000>=1.31,<2
duckdb>=1.0

# Browser Automation Runtime — structured browser control behind Eclipse's own
# provider/runtime abstraction (the model never sees a raw Playwright object).
# OPTIONAL in effect: a missing Playwright makes the browser toolset report an
# honest BROWSER_UNAVAILABLE (it is never registered), exactly like the
# 503-runtime-unavailable contract. Installed by default so `browser_*` works
# out of the box. The browser BINARY is a separate, one-time download
# (`python -m playwright install chromium`); the provider reports
# BROWSER_UNINSTALLED until it is present.
# Pinned to the 1.62 series so it shares the SAME chromium build the app's Node
# Playwright already uses (they cache under one ms-playwright directory).
playwright>=1.62,<1.63

# Computer Use Engine (native desktop) — OPTIONAL in effect. Every backend here
# is honest: a missing one degrades to UNAVAILABLE/reason, never a fake desktop.
# * Pillow: image capture (GDI fallback encoding) + vision frame encoding.
# * uiautomation (Windows): the rich accessibility tree (UIA). Without it the
#   driver falls back to the Win32 child-window baseline.
# * winsdk (Windows): the OS-native OCR engine (Windows.Media.Ocr) — no external
#   binary, no download, no cloud.
# Cross-platform Pillow is installed by default; the Windows-only backends carry
# markers so they are not attempted on macOS/Linux.
Pillow>=10.0
uiautomation>=2.0; sys_platform == "win32"
winsdk>=1.0.0b10; sys_platform == "win32"

# Document Authoring — native creation of Office documents (Word .docx, Excel
# .xlsx, PowerPoint .pptx), the formats Document Intelligence can READ but not
# write. Three small, pure-Python OOXML writers; nothing here executes a macro or
# runs an Office runtime — they build the ZIP+XML container directly. OPTIONAL in
# effect: a missing library makes `document_create` report an honest
# DOCUMENT_LIBRARY_MISSING for that format rather than emitting a fake file, so a
# partial install degrades one format at a time. Installed by default so Word /
# Excel / PowerPoint creation works out of the box.
# * python-docx  — WordprocessingML
# * openpyxl     — SpreadsheetML
# * python-pptx  — PresentationML (pulls lxml/Pillow/XlsxWriter, all present)
python-docx>=1.1,<2
openpyxl>=3.1,<4
python-pptx>=0.6

pypdfium2>=4
